1)Image Filtering and Enhancement: Apply convolution, edge detection (Sobel, Canny), blurring, and histogram equalization to enhance and analyze grayscale/color images.



import cv2
import numpy as np
import matplotlib.pyplot as plt
import os

# Path to image
img_path = 'dog1.jpg'

# --- Read + basic check ---
image_bgr = cv2.imread(img_path)
if image_bgr is None:
    raise FileNotFoundError(f"Cannot read image at '{img_path}'. Check path and filename.")

# Keep both BGR (for OpenCV ops) and RGB (for plotting)
image_rgb = cv2.cvtColor(image_bgr, cv2.COLOR_BGR2RGB)
image_gray = cv2.cvtColor(image_bgr, cv2.COLOR_BGR2GRAY)  # read from BGR -> gray

# --- Define kernel (sharpen) ---
kernel = np.array([[0, -1, 0],
                   [-1, 5, -1],
                   [0, -1, 0]], dtype=np.float32)

# Helper to ensure uint8 output for visualization
def to_uint8(img):
    if img.dtype == np.uint8:
        return img
    img = np.clip(img, 0, 255)
    return img.astype(np.uint8)

# --- Processing on Grayscale Image ---
# 1) Convolution (Sharpen) on grayscale
convolved_gray = cv2.filter2D(image_gray, ddepth=-1, kernel=kernel)
convolved_gray = to_uint8(convolved_gray)

# 2) Sobel Edge Detection (use convertScaleAbs for proper uint8)
sobel_x = cv2.Sobel(image_gray, cv2.CV_64F, 1, 0, ksize=3)
sobel_y = cv2.Sobel(image_gray, cv2.CV_64F, 0, 1, ksize=3)
sobel_mag = cv2.magnitude(sobel_x, sobel_y)                 # float64
sobel_combined_gray = cv2.convertScaleAbs(sobel_mag)        # uint8 scaled

# 3) Canny Edge Detection
canny_edges_gray = cv2.Canny(image_gray, 100, 200)

# 4) Gaussian Blur
blurred_gray = cv2.GaussianBlur(image_gray, (7, 7), 0)

# 5) Histogram Equalization (grayscale)
hist_eq_gray = cv2.equalizeHist(image_gray)

# --- Processing on Color (BGR) Image ---
# 1) Convolution (Sharpen) on each channel by applying filter2D on BGR
convolved_bgr = cv2.filter2D(image_bgr, ddepth=-1, kernel=kernel)
convolved_rgb = cv2.cvtColor(to_uint8(convolved_bgr), cv2.COLOR_BGR2RGB)

# 2) Gaussian Blur on color
blurred_bgr = cv2.GaussianBlur(image_bgr, (7, 7), 0)
blurred_rgb = cv2.cvtColor(blurred_bgr, cv2.COLOR_BGR2RGB)

# 3) Histogram equalization on luminance (Y) channel using YUV
image_yuv = cv2.cvtColor(image_bgr, cv2.COLOR_BGR2YUV)
image_yuv[:, :, 0] = cv2.equalizeHist(image_yuv[:, :, 0])
hist_eq_bgr = cv2.cvtColor(image_yuv, cv2.COLOR_YUV2BGR)
hist_eq_rgb = cv2.cvtColor(hist_eq_bgr, cv2.COLOR_BGR2RGB)

# Optional: save outputs (uncomment if you want files)
out_dir = "outputs"
os.makedirs(out_dir, exist_ok=True)
cv2.imwrite(os.path.join(out_dir, "convolved_gray.png"), convolved_gray)
cv2.imwrite(os.path.join(out_dir, "sobel_gray.png"), sobel_combined_gray)
cv2.imwrite(os.path.join(out_dir, "canny_gray.png"), canny_edges_gray)
cv2.imwrite(os.path.join(out_dir, "blurred_gray.png"), blurred_gray)
cv2.imwrite(os.path.join(out_dir, "hist_eq_gray.png"), hist_eq_gray)
cv2.imwrite(os.path.join(out_dir, "convolved_rgb.png"), cv2.cvtColor(convolved_rgb, cv2.COLOR_RGB2BGR))
cv2.imwrite(os.path.join(out_dir, "blurred_rgb.png"), cv2.cvtColor(blurred_rgb, cv2.COLOR_RGB2BGR))
cv2.imwrite(os.path.join(out_dir, "hist_eq_rgb.png"), cv2.cvtColor(hist_eq_rgb, cv2.COLOR_RGB2BGR))

# --- Plotting ---
fig, axs = plt.subplots(4, 3, figsize=(16, 18))

# Row 1: Originals
axs[0, 0].imshow(image_gray, cmap='gray')
axs[0, 0].set_title("Original Grayscale")
axs[0, 1].imshow(image_rgb)
axs[0, 1].set_title("Original RGB")
axs[0, 2].axis('off')

# Row 2: Convolution (Sharpen)
axs[1, 0].imshow(convolved_gray, cmap='gray')
axs[1, 0].set_title("Grayscale Sharpened")
axs[1, 1].imshow(convolved_rgb)
axs[1, 1].set_title("RGB Sharpened")
axs[1, 2].axis('off')

# Row 3: Edge Detection (grayscale)
axs[2, 0].imshow(sobel_combined_gray, cmap='gray')
axs[2, 0].set_title("Grayscale Sobel (magnitude)")
axs[2, 1].imshow(canny_edges_gray, cmap='gray')
axs[2, 1].set_title("Grayscale Canny")
axs[2, 2].axis('off')

# Row 4: Blur and Histogram Equalization
axs[3, 0].imshow(blurred_gray, cmap='gray')
axs[3, 0].set_title("Grayscale Gaussian Blur")
axs[3, 1].imshow(blurred_rgb)
axs[3, 1].set_title("RGB Gaussian Blur")
axs[3, 2].imshow(hist_eq_rgb)
axs[3, 2].set_title("RGB Histogram Equalization (YUV -> Y eq)")
# Turn off axes for all
for ax in axs.ravel():
    ax.axis('off')

2)Color Space Transformations: Convert images between RGB, HSV, Lab color spaces and extract features for segmentation using k-means clustering.

import cv2
import numpy as np
import matplotlib.pyplot as plt

# Load image in RGB
image_bgr = cv2.imread('dog1.jpg')
image_rgb = cv2.cvtColor(image_bgr, cv2.COLOR_BGR2RGB)

# Convert RGB to HSV
image_hsv = cv2.cvtColor(image_rgb, cv2.COLOR_RGB2HSV)

# Convert RGB to Lab
image_lab = cv2.cvtColor(image_rgb, cv2.COLOR_RGB2Lab)

# Flatten the image pixels for clustering (convert to float32 for kmeans)
pixel_values = image_rgb.reshape((-1, 3))
pixel_values = np.float32(pixel_values)

# Define criteria and apply kmeans clustering
criteria = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 100, 0.2)
k = 4  # number of clusters
_, labels, centers = cv2.kmeans(pixel_values, k, None, criteria, 10, cv2.KMEANS_RANDOM_CENTERS)

# Convert centers back to uint8 (color values) and reshape labels to image shape
centers = np.uint8(centers)
segmented_image = centers[labels.flatten()]
segmented_image = segmented_image.reshape(image_rgb.shape)

# Plot results
fig, axs = plt.subplots(2, 2, figsize=(12, 10))

axs[0, 0].imshow(image_rgb)
axs[0, 0].set_title('Original RGB Image')
axs[0, 0].axis('off')

axs[0, 1].imshow(image_hsv)
axs[0, 1].set_title('HSV Image')
axs[0, 1].axis('off')

axs[1, 0].imshow(image_lab)
axs[1, 0].set_title('Lab Image')
axs[1, 0].axis('off')

axs[1, 1].imshow(segmented_image)
axs[1, 1].set_title(f'Segmented Image with {k} clusters')

3)CNN Basics for Image Classification: Build a simple CNN from scratch using PyTorch to classify images from the CIFAR-10 dataset. Train and evaluate the model.

import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset

# -----------------------------
# 1. Load CIFAR-10 dataset (.npz)
# -----------------------------
data = np.load("cifar-10.npz")
x_train, y_train = data['x_train'], data['y_train']
x_test, y_test = data['x_test'], data['y_test']

# Normalize and convert to torch tensors
x_train = torch.tensor(x_train / 255.0, dtype=torch.float32).permute(0, 3, 1, 2)
x_test = torch.tensor(x_test / 255.0, dtype=torch.float32).permute(0, 3, 1, 2)
y_train = torch.tensor(y_train, dtype=torch.long).squeeze()
y_test = torch.tensor(y_test, dtype=torch.long).squeeze()

# Dataloaders
train_loader = DataLoader(TensorDataset(x_train, y_train), batch_size=64, shuffle=True)
test_loader = DataLoader(TensorDataset(x_test, y_test), batch_size=64, shuffle=False)

# -----------------------------
# 2. Define a Simple CNN Model
# -----------------------------
class SimpleCNN(nn.Module):
    def __init__(self):
        super(SimpleCNN, self).__init__()
        self.conv_layers = nn.Sequential(
            nn.Conv2d(3, 16, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2, 2),
            nn.Conv2d(16, 32, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2, 2)
        )
        self.fc_layers = nn.Sequential(
            nn.Flatten(),
            nn.Linear(32 * 8 * 8, 128),
            nn.ReLU(),
            nn.Linear(128, 10)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.fc_layers(x)
        return x

# -----------------------------
# 3. Train and Evaluate
# -----------------------------
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model = SimpleCNN().to(device)
criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=0.001)

# Training
for epoch in range(5):  # small epochs for simplicity
    model.train()
    total_loss = 0
    for images, labels in train_loader:
        images, labels = images.to(device), labels.to(device)
        optimizer.zero_grad()
        outputs = model(images)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        total_loss += loss.item()
    print(f"Epoch [{epoch+1}/5], Loss: {total_loss/len(train_loader):.4f}")

# Evaluation
model.eval()
correct, total = 0, 0
with torch.no_grad():
    for images, labels in test_loader:
        images, labels = images.to(device), labels.to(device)
        outputs = model(images)
        _, predicted = torch.max(outputs, 1)
        total += labels.size(0)
        correct += (predicted == labels).sum().item()

print(f"Test Accuracy: {100 * correct / total:.2f}%")

5)Comparing Classical vs. Deep Features: Extract SIFT, ORB features and compare with CNN feature embeddings for matching similar images in a dataset.(dog)

# 1. Import necessary libraries
import cv2
import numpy as np
import torch
from sklearn.metrics.pairwise import cosine_similarity
from torchvision import models, transforms
from PIL import Image
import matplotlib.pyplot as plt

# 2. Load and preprocess images
def load_image(image_path, target_size=(224, 224)):
    """
    Loads an image, resizes it to the target size, and converts it to grayscale.
    """
    img = cv2.imread(image_path)
    img_resized = cv2.resize(img, target_size)  # Resize to target size
    return cv2.cvtColor(img_resized, cv2.COLOR_BGR2GRAY), img_resized

# 3. Extract Classical Features (SIFT and ORB)
def extract_sift_features(image):
    """
    Extracts keypoints and descriptors using SIFT.
    """
    sift = cv2.SIFT_create()
    keypoints, descriptors = sift.detectAndCompute(image, None)
    return keypoints, descriptors

def extract_orb_features(image):
    """
    Extracts keypoints and descriptors using ORB.
    """
    orb = cv2.ORB_create()
    keypoints, descriptors = orb.detectAndCompute(image, None)
    return keypoints, descriptors

# 4. Extract Deep Learning Features (CNN Embeddings)
def extract_cnn_embeddings(image_path):
    """
    Extracts feature embeddings from a pre-trained CNN (ResNet50).
    """
    # Load pre-trained model (ResNet50)
    model = models.resnet50(pretrained=True)
    model.eval()

    # Preprocess the image for ResNet50
    preprocess = transforms.Compose([
        transforms.Resize(256),
        transforms.CenterCrop(224),
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
    ])

    # Load and preprocess image
    image = Image.open(image_path)
    image = preprocess(image).unsqueeze(0)

    # Extract features from the second-to-last layer (penultimate layer)
    with torch.no_grad():
        features = model(image)
    
    return features.numpy().flatten()

# 5. Compare Features
def compare_sift_orb(descriptors1, descriptors2):
    """
    Compares two sets of descriptors (SIFT/ORB) using BFMatcher (Brute-Force Matcher).
    """
    bf = cv2.BFMatcher(cv2.NORM_L2, crossCheck=True)
    matches = bf.match(descriptors1, descriptors2)
    return matches

def compare_cnn_embeddings(embedding1, embedding2):
    """
    Compares two CNN embeddings using cosine similarity.
    """
    return cosine_similarity([embedding1], [embedding2])

# 6. Visualize Matches (for SIFT and ORB)
def visualize_matches(image1, image2, keypoints1, keypoints2, matches, title="Feature Matches"):
    """
    Visualizes feature matches between two images.
    """
    img_matches = cv2.drawMatches(image1, keypoints1, image2, keypoints2, matches, None, flags=cv2.DrawMatchesFlags_NOT_DRAW_SINGLE_POINTS)
    plt.figure(figsize=(12, 6))
    plt.imshow(img_matches)
    plt.title(title)
    plt.axis('off')
    plt.show()

# 7. Example usage with images
image_path1 = 'dog1.jpg'  # Replace with your image path
image_path2 = 'dog2.jpg'  # Replace with your image path

# Resize target size
target_size = (224, 224)

# Load and resize images
image1_gray, image1_resized = load_image(image_path1, target_size)
image2_gray, image2_resized = load_image(image_path2, target_size)

# Step 1: Extract classical features (SIFT and ORB)
keypoints1_sift, descriptors1_sift = extract_sift_features(image1_gray)
keypoints2_sift, descriptors2_sift = extract_sift_features(image2_gray)

keypoints1_orb, descriptors1_orb = extract_orb_features(image1_gray)
keypoints2_orb, descriptors2_orb = extract_orb_features(image2_gray)

# Step 2: Extract deep learning features (CNN embeddings)
embedding1_cnn = extract_cnn_embeddings(image_path1)
embedding2_cnn = extract_cnn_embeddings(image_path2)

# Step 3: Compare classical features (SIFT)
matches_sift = compare_sift_orb(descriptors1_sift, descriptors2_sift)

# Step 4: Compare classical features (ORB)
matches_orb = compare_sift_orb(descriptors1_orb, descriptors2_orb)

# Step 5: Compare deep features (CNN embeddings)
similarity_cnn = compare_cnn_embeddings(embedding1_cnn, embedding2_cnn)

# Step 6: Display the results
print(f"SIFT matches: {len(matches_sift)}")
print(f"ORB matches: {len(matches_orb)}")
print(f"Cosine similarity (CNN embeddings): {similarity_cnn[0][0]:.4f}")

# Visualize matches for SIFT
visualize_matches(image1_resized, image2_resized, keypoints1_sift, keypoints2_sift, matches_sift, title="SIFT Matches")

# Visualize matches for ORB
visualize_matches(image1_resized, image2_resized, keypoints1_orb, keypoints2_orb, matches_orb, title="ORB Matches")

def extract_class_from_name(path: str) -> str:
    """Infer class label from filename prefix before first non-letter."""
    stem = Path(path).stem.lower()
    cls = ""
    for ch in stem:
        if ch.isalpha():
            cls += ch
        else:
            break
    return cls or stem  # fallback


def sample_indices(num_total, num_frames):
    if num_total <= 0:
        return [0] * num_frames
    if num_total <= num_frames:
        idxs = list(range(num_total))
        while len(idxs) < num_frames:
            idxs.append(num_total - 1)
        return idxs
    step = num_total / float(num_frames)
    return [min(num_total - 1, int(i * step)) for i in range(num_frames)]


def read_video_cv2(path):
    """Reads video using OpenCV → returns (T,H,W,C) uint8 RGB."""
    cap = cv2.VideoCapture(path)
    frames = []
    while True:
        ok, frame = cap.read()
        if not ok:
            break
        frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
        frames.append(frame)
    cap.release()
    return np.array(frames, dtype=np.uint8)

class VideoActionDataset(Dataset):
    def __init__(self, root, num_frames=16):
        self.samples = []
        for p in sorted(Path(root).glob("*")):
            if p.suffix.lower() in [".mp4", ".avi", ".mov", ".mkv"]:
                self.samples.append(str(p))

        if len(self.samples) == 0:
            raise RuntimeError(f"No videos found in {root}")

        labels = [extract_class_from_name(p) for p in self.samples]
        classes = sorted(set(labels))
        self.class_to_idx = {c: i for i, c in enumerate(classes)}
        self.idx_to_class = {i: c for c, i in self.class_to_idx.items()}
        self.labels = [self.class_to_idx[l] for l in labels]
        self.num_frames = num_frames

        print("Found videos:")
        for p, lbl in zip(self.samples, self.labels):
            print(f"  {p} -> class '{self.idx_to_class[lbl]}'")
        print("Classes:", self.class_to_idx)

    def __len__(self):
        return len(self.samples)

    def __getitem__(self, idx):
        path = self.samples[idx]
        label = self.labels[idx]

        # --- Load video using OpenCV ---
        video = read_video_cv2(path)  # (T,H,W,C)
        T_total = video.shape[0]

        idxs = sample_indices(T_total, self.num_frames)
        clip = video[idxs]  # (T,H,W,C)

        clip = torch.from_numpy(clip).float() / 255.0

        return clip, label

class MiniI3D(nn.Module):
    def __init__(self, in_ch=3, num_classes=2):
        super().__init__()
        self.conv1 = nn.Conv3d(in_ch, 16, 3, padding=1)
        self.pool1 = nn.MaxPool3d((1,2,2))
        self.conv2 = nn.Conv3d(16, 32, 3, padding=1)
        self.pool2 = nn.MaxPool3d((2,2,2))
        self.conv3 = nn.Conv3d(32, 64, 3, padding=1)
        self.pool3 = nn.AdaptiveAvgPool3d((1,1,1))
        self.fc = nn.Linear(64, num_classes)

    def forward(self, x):
        # x: (B,C,T,H,W)
        x = F.relu(self.conv1(x))
        x = self.pool1(x)
        x = F.relu(self.conv2(x))
        x = self.pool2(x)
        x = F.relu(self.conv3(x))
        x = self.pool3(x)
        x = x.view(x.size(0), -1)
        return self.fc(x)

def run_epoch(loader, training=True):
    if training:
        model.train()
    else:
        model.eval()
    total_loss, total_correct, total_count = 0.0, 0, 0

    for clips, labels in loader:
        clips = clips.to(DEVICE)
        labels = labels.to(DEVICE)

        if training:
            optimizer.zero_grad()

        logits = model(clips)
        loss = criterion(logits, labels)

        if training:
            loss.backward()
            optimizer.step()

        total_loss += loss.item() * clips.size(0)
        preds = logits.argmax(1)
        total_correct += (preds == labels).sum().item()
        total_count += clips.size(0)

    return total_loss / max(1,total_count), total_correct / max(1,total_count)

# ------------------------------
# TEMPORAL FEATURE ANALYSIS (Grad-CAM style)
# ------------------------------
def temporal_saliency(model, clip, target_class=None):
    """
    clip: (1,C,T,H,W)
    returns: saliency list for each frame
    """
    model.eval()
    clip = clip.to(DEVICE).requires_grad_(True)

    conv_layer = model.conv2
    activations = {}
    gradients = {}

    def fwd_hook(m, inp, out):
        activations["value"] = out

    def bwd_hook(m, gin, gout):
        gradients["value"] = gout[0]

    h1 = conv_layer.register_forward_hook(fwd_hook)
    h2 = conv_layer.register_backward_hook(bwd_hook)

    logits = model(clip)
    if target_class is None:
        target = logits.argmax(1).item()
    else:
        target = target_class

    score = logits[0, target]
    model.zero_grad()
    score.backward()

    act = activations["value"][0]   # (C,T,H,W)
    grad = gradients["value"][0]    # (C,T,H,W)

    # average gradient over spatial dims -> (C,T)
    weights = grad.mean(dim=(2, 3))

    # weighted sum across channels -> (T,H,W)
    weighted = (weights.unsqueeze(2).unsqueeze(3) * act).sum(dim=0)
    saliency = weighted.mean(dim=(1,2))  # (T,)

    saliency = saliency - saliency.min()
    if saliency.max() > 0:
        saliency = saliency / saliency.max()

    h1.remove()
    h2.remove()

    return saliency.detach().cpu().tolist(), target

#!/usr/bin/env python3

import os
os.environ["HF_HUB_OFFLINE"] = "1"   # Force offline mode

import torch
from diffusers import AutoPipelineForText2Image
from PIL import Image
import matplotlib.pyplot as plt

def main():
    print("PyTorch:", torch.__version__)
    print("CUDA available:", torch.cuda.is_available())
    device = "cuda" if torch.cuda.is_available() else "cpu"
    print("Using device:", device)

    model_path = "./sd-turbo-local"   # local model folder

    if not os.path.isdir(model_path):
        raise FileNotFoundError(
            f"Local model folder not found: {model_path}\n"
            "Please place the sd-turbo pipeline folder here or set `model_path` to the correct location."
        )
    print("✓ Found local model folder:", model_path)

    torch_dtype = torch.float16 if torch.cuda.is_available() else torch.float32

    print("Loading pipeline from local files (offline)...")
    pipe = AutoPipelineForText2Image.from_pretrained(
        model_path,
        torch_dtype=torch_dtype,
        local_files_only=True,
    )

    pipe = pipe.to(device)

    if device == "cuda":
        try:
            pipe.enable_attention_slicing()
        except:
            pass

    print("✓ Text-to-image pipeline loaded from local disk.")

    # ---------------------------------------------------
    # Image generation helper
    # ---------------------------------------------------
    def generate_image(prompt, num_inference_steps=4, guidance_scale=0.0, seed=42, out_file=None):
        gen = torch.Generator(device=device).manual_seed(seed)

        print(f"\nPrompt: {prompt}")
        result = pipe(
            prompt=prompt,
            num_inference_steps=num_inference_steps,
            guidance_scale=guidance_scale,
            generator=gen,
        )

        img = result.images[0]

        plt.figure(figsize=(6, 6))
        plt.imshow(img)
        plt.axis("off")
        plt.title(prompt[:80], fontsize=10)
        plt.show()

        if out_file:
            img.save(out_file)
            print("Saved image to:", out_file)

        return img

    # ---------------------------------------------------
    # Example prompt (you can change this)
    # ---------------------------------------------------
    prompt_basic = (
        "a giant ancient temple floating in the clouds, golden sunlight, "
        "mystical energy, fantasy landscape, breathtaking atmosphere"
    )

    out = generate_image(prompt_basic, num_inference_steps=4, guidance_scale=0.0, seed=1, out_file="sd_turbo_output.png")
    print("Done.")

if __name__ == "__main__":
    main()
